25 // NOTE: On Windows, the DLL exports its API (KRONKNET_BUILD_DLL, set when it is built),
26 // and who links with it can import it (KRONKNET_USE_DLL, set by the kronknet-shared target).
27 // Without either, nothing is needed: a static library needs none, and the API is only
28 // functions, which a DLL gives through its import library anyway
29 #if defined(_WIN32) || defined(__CYGWIN__)
30 #if defined(KRONKNET_BUILD_DLL)
31 #define KN_API __declspec(dllexport)
32 #elif defined(KRONKNET_USE_DLL)
33 #define KN_API __declspec(dllimport)
34 #else
35 #define KN_API
36 #endif
